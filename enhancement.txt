    uint32_t size_class_frequency[16];      // How often each size used

void lgx_learn_allocation_pattern(lgx_pattern_learner_t* learner, 
                                   lgx_allocation_intent_t* intent) {
    // Update moving averages
    learner->allocation_history[learner->allocation_count % 1000] = intent->size;
    learner->size_class_frequency[size_to_class(intent->size)]++;
    
    // Exponential smoothing (simple, effective)
    double alpha = 0.1;  // Smoothing factor
    learner->exponential_moving_avg[size_to_class(intent->size)] = 
        alpha * intent->size + (1 - alpha) * learner->exponential_moving_avg[...];

## The "Predictive Optimization" Vision: Practical Implementation

### **Your Vision**:
```
Predict allocations 2 frames ahead, eliminate stalls
```

### **Layer 2 Implementation** (No ML):

```c
// Simple prediction via exponential smoothing

typedef struct lgx_frame_predictor {
    double last_allocation_size;
    double predicted_next_size;
    uint32_t confidence;  // How often prediction was correct
} lgx_frame_predictor_t;

void lgx_predict_next_frame(lgx_frame_predictor_t* pred) {
    // Exponential smoothing (70% history, 30% latest)
    pred->predicted_next_size = 
        0.7 * pred->predicted_next_size + 
        0.3 * pred->last_allocation_size;
    
    // Pre-warm pool if confidence > 90%
    if (pred->confidence > 90) {
        prewarm_pool(pred->predicted_next_size);
    }
}

// Update after each frame
void lgx_end_frame(lgx_frame_predictor_t* pred, double actual_size) {
    double error = fabs(pred->predicted_next_size - actual_size);
    
    if (error < 0.1 * actual_size) {  // Within 10%
        pred->confidence++;
    } else {
        pred->confidence = max(0, pred->confidence - 10);  // Decay faster on miss
    }
    
    pred->last_allocation_size = actual_size;
}
```

**Complexity**: ~200 lines of code, not 20,000 lines of ML infrastructure.

**Effectiveness**: Empirically, exponential smoothing achieves 80-90% accuracy for repetitive workloads (games are repetitive).

**Future**: If this works, **then** add ML (Phase 5+). If it doesn't, you've wasted 1 month, not 18.
